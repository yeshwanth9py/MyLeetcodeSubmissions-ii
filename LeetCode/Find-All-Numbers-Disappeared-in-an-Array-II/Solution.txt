1class Solution {
2public:
3    vector<vector<int>> findDisappearedNumbers(vector<int>& arr, int lo, int hi) {
4        sort(arr.begin(), arr.end());
5        if(arr.back()<hi){
6            arr.push_back(hi+1);
7        }
8        arr.erase(unique(arr.begin(), arr.end()), arr.end());
9        int n = arr.size();
10        int i = lower_bound(arr.begin(), arr.end(), lo) - arr.begin();
11        if(i == n){
12            return {{lo, hi}};
13        }
14        vector<vector<int>> ans;
15        if(arr[i] != lo){
16            ans.push_back({lo, min(arr[i]-1, hi)});
17            if(arr[i]>=hi) return ans;
18        }
19        i++;
20        while(i<n){
21            if(arr[i] == arr[i-1]+1){
22                i++;
23            }else{
24                if(arr[i-1]+1>hi) break;
25                ans.push_back({arr[i-1]+1, min(arr[i]-1, hi)});
26                if(arr[i]-1>=hi) break;
27                i++;
28            }
29        }
30
31        
32
33        return ans;
34    }
35};
