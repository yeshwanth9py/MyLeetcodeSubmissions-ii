1class Solution {
2public:
3    int maxValidSplits(vector<int>& nums) {
4        int n = nums.size();
5        if(n == 1) return 0;
6        
7        vector<int> gcdd(n, 0);
8        int g = 0;
9        for(int i=0; i<n; i++){
10            g = __gcd(nums[i], g);
11            gcdd[i] = g;
12        }
13
14        g = 0;
15        int ans = 0;
16        vector<int> extra;
17        
18        for(int i=n-1; i>0; i--){
19            g = __gcd(g, nums[i]);
20            if(g == gcdd[i-1]){
21                ans++;
22            }else{
23                extra.push_back(i-1);
24            }
25        }
26
27        vector<vector<int>> dpg(n, vector<int>(n, 0));
28        vector<vector<int>> dpg2(n, vector<int>(n, 0));
29
30        for(int i=0; i<n; i++){
31            int st = nums[i];
32            nums[i] = 0;
33            int g = 0;
34            for(int j=0; j<n; j++){
35                g = __gcd(g, nums[j]);
36                dpg[i][j] = g;
37            }
38
39            g = 0;
40            for(int j=n-1; j>=0; j--){
41                g = __gcd(g, nums[j]);
42                dpg2[i][j] = g;
43            }
44            
45            nums[i] = st;
46        }
47
48        for(int idx: extra){
49            for(int i=0; i<n; i++){
50                if(i == idx) continue; 
51                if(dpg[i][idx] == dpg2[i][idx+1]){
52                    ans++;
53                }
54            }
55        }
56
57        return ans;
58    }
