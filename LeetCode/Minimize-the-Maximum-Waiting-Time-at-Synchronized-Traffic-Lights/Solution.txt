1class Solution {
2public:
3    int minPenalty(int p, vector<int>& larr, vector<int>& arr) {
4        sort(larr.begin(), larr.end());
5        int ans = 0;
6        for(int i=0; i<arr.size(); i++){
7            arr[i] = arr[i]%p;
8            auto it = upper_bound(larr.begin(), larr.end(), arr[i]);
9            if(it == larr.end()){
10                ans = max(ans, p-arr[i]);
11                continue;
12            }
13        }
14        return ans;
15    }
16};
